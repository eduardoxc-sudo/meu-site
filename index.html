function gerarSemanaAutomatica(chaveSemana) { if (!estado.disciplinas || estado.disciplinas.length === 0) { estado.semanas[chaveSemana] = {}; salvarEstado(); return; } estado.porDia = parseInt(document.getElementById('materiasPorDia').value) || 2; let pool = []; estado.disciplinas.forEach(d => { for (let i = 0; i < d.peso; i++) pool.push(d.nome); }); for (let i = pool.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [pool[i], pool[j]] = [pool[j], pool[i]]; } const semanaExistente = estado.semanas[chaveSemana] || {}; estado.semanas[chaveSemana] = {}; let ponteiro = estado.ponteiroGeral || 0; // Obtém a data de hoje sem horas para comparação estrita const hoje = new Date(); hoje.setHours(0, 0, 0, 0); const segunda = obterSegundaFeira(dataReferencia); const diasComData = obterDiasDaSemana(segunda); diasComData.forEach((d, diaIdx) => { const slotsAntigos = semanaExistente[d.nome] || []; estado.semanas[chaveSemana][d.nome] = []; // Compara a data do slot com hoje const dataDoSlot = new Date(d.dataCompleta); dataDoSlot.setHours(0, 0, 0, 0); // Se o dia já passou e não tem nada concluído nele, deixa vazio if (dataDoSlot < hoje) { // Mantém apenas o que você porventura já concluiu anteriormente const concluidos = slotsAntigos.filter(s => s.feito); estado.semanas[chaveSemana][d.nome] = concluidos; return; } for (let k = 0; k < estado.porDia; k++) { if (slotsAntigos[k] && slotsAntigos[k].feito) { estado.semanas[chaveSemana][d.nome].push(slotsAntigos[k]); continue; } const nomeMateria = pool[ponteiro % pool.length]; const infoTopico = obterProximoTopicoPrioritario(nomeMateria); estado.semanas[chaveSemana][d.nome].push({ id: 'slot_' + Math.random().toString(36).substr(2, 9), materia: nomeMateria, assunto: infoTopico.titulo, prioridade: infoTopico.prioridade, feito: false }); ponteiro++; } }); estado.ponteiroGeral = ponteiro; salvarEstado(); }
